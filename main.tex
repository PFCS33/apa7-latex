% ---------------------------------------------------------------------
%  Document mode -- change "stu" to switch the layout:

% ---------------------------------------------------------------------
%  Font: Times New Roman 12 pt (requires XeLaTeX, set in .latexmkrc).

  \setmainfont{Times New Roman}%
}{%

    Extension      = .otf,
    UprightFont    = *-regular,
    BoldFont       = *-bold,
    ItalicFont     = *-italic,
    BoldItalicFont = *-bolditalic
  ]%
}
% Times New Roman has no math symbols; use the matching Times-style math font.
\setmathfont{texgyretermes-math.otf}
